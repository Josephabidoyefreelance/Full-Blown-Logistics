import { randomUUID } from 'node:crypto';
import { NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';
import { JAAD_PLANS, type JaadPlanCode } from '@/lib/billing/plans';
import { createBillingAdminClient } from '@/lib/billing/flutterwave';

export const runtime = 'nodejs';

function isPlanCode(value: unknown): value is JaadPlanCode {
  return typeof value === 'string' && Object.prototype.hasOwnProperty.call(JAAD_PLANS, value);
}

export async function POST(request: Request) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Please sign in to manage your company package.' }, { status: 401 });

  const body = await request.json().catch(() => ({})) as { planCode?: unknown };
  if (!isPlanCode(body.planCode)) {
    return NextResponse.json({ error: 'Choose a valid JAAD package.' }, { status: 400 });
  }
  const plan = JAAD_PLANS[body.planCode];

  const { data: profile, error: profileError } = await supabase
    .from('profiles')
    .select('id, email, company_id')
    .eq('id', user.id)
    .maybeSingle();
  if (profileError || !profile?.company_id) {
    return NextResponse.json({ error: 'Your account is not assigned to a company workspace. Contact your administrator.' }, { status: 409 });
  }
  const email = profile.email || user.email;
  if (!email) return NextResponse.json({ error: 'A valid billing email is required.' }, { status: 400 });

  let admin;
  try {
    admin = createBillingAdminClient();
  } catch {
    return NextResponse.json({ error: 'Billing is not configured on the server yet.' }, { status: 503 });
  }

  const { data: company, error: companyError } = await admin
    .from('companies')
    .select('id, name, owner_user_id')
    .eq('id', profile.company_id)
    .maybeSingle();
  if (companyError || !company) return NextResponse.json({ error: 'Your company workspace could not be found.' }, { status: 404 });
  if (company.owner_user_id !== user.id) {
    return NextResponse.json({ error: 'Only the company billing owner can choose or renew a package.' }, { status: 403 });
  }

  const reference = `JAAD-${randomUUID().replaceAll('-', '')}`;
  const amountKobo = plan.priceNaira * 100;
  const { data: attempt, error: insertError } = await admin
    .from('billing_payment_attempts')
    .insert({
      company_id: company.id,
      initiated_by: user.id,
      plan_code: plan.code,
      amount_kobo: amountKobo,
      currency: 'NGN',
      reference,
      status: 'pending',
      payment_provider: 'flutterwave',
    })
    .select('id')
    .single();
  if (insertError || !attempt) {
    console.error('JAAD billing attempt insert failed:', insertError?.message);
    return NextResponse.json({ error: 'Could not prepare the bank transfer request.' }, { status: 500 });
  }

  const appUrl = process.env.APP_URL;
  const secret = process.env.FLW_SECRET_KEY;
  if (!appUrl || !secret) {
    await admin.from('billing_payment_attempts').update({ status: 'failed', failure_message: 'Billing environment is incomplete.' }).eq('id', attempt.id);
    return NextResponse.json({ error: 'Set APP_URL and FLW_SECRET_KEY in the server environment.' }, { status: 503 });
  }

  try {
    const flutterwaveResponse = await fetch('https://api.flutterwave.com/v3/payments', {
      method: 'POST',
      headers: { Authorization: `Bearer ${secret}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        tx_ref: reference,
        amount: plan.priceNaira,
        currency: 'NGN',
        redirect_url: new URL('/api/billing/callback', appUrl).toString(),
        payment_options: 'banktransfer',
        customer: { email, name: company.name },
        meta: { payment_attempt_id: attempt.id, company_id: company.id, plan_code: plan.code },
        customizations: { title: 'JAAD Logistics ERP', description: `${plan.name} subscription` },
      }),
      cache: 'no-store',
    });
    const result = await flutterwaveResponse.json().catch(() => null) as
      | { status?: string; message?: string; data?: { link?: string } } | null;
    const checkoutUrl = result?.data?.link;
    if (!flutterwaveResponse.ok || result?.status !== 'success' || !checkoutUrl) {
      await admin.from('billing_payment_attempts')
        .update({ status: 'failed', failure_message: result?.message || 'Flutterwave checkout initialization failed.' })
        .eq('id', attempt.id);
      return NextResponse.json({ error: 'Flutterwave could not start bank-transfer checkout. Check your Flutterwave account payment-method settings.' }, { status: 502 });
    }

    await admin.from('billing_payment_attempts').update({ checkout_url: checkoutUrl }).eq('id', attempt.id);
    return NextResponse.json({ authorizationUrl: checkoutUrl });
  } catch (error) {
    await admin.from('billing_payment_attempts')
      .update({ status: 'failed', failure_message: 'Unable to reach Flutterwave.' })
      .eq('id', attempt.id);
    console.error('JAAD Flutterwave checkout network error:', error instanceof Error ? error.message : 'unknown error');
    return NextResponse.json({ error: 'Unable to reach Flutterwave. Please retry shortly.' }, { status: 502 });
  }
}

import { NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';
import { verifyAndActivateBankTransfer } from '@/lib/billing/flutterwave';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

function packageRedirect(status: 'success' | 'pending' | 'failed') {
  const appUrl = process.env.APP_URL;
  if (!appUrl) throw new Error('APP_URL is not configured.');
  const url = new URL('/packages', appUrl);
  url.searchParams.set('payment', status);
  return NextResponse.redirect(url);
}

export async function GET(request: Request) {
  const url = new URL(request.url);
  const reference = url.searchParams.get('tx_ref');
  if (!reference) return packageRedirect('failed');

  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) {
    const appUrl = process.env.APP_URL;
    if (!appUrl) return NextResponse.json({ error: 'APP_URL is not configured.' }, { status: 503 });
    const login = new URL('/admin/login', appUrl);
    login.searchParams.set('next', '/packages');
    return NextResponse.redirect(login);
  }

  const { data: profile } = await supabase.from('profiles').select('company_id').eq('id', user.id).maybeSingle();
  if (!profile?.company_id) return packageRedirect('failed');

  const result = await verifyAndActivateBankTransfer(reference, profile.company_id, url.searchParams.get('transaction_id') || undefined);
  if (result.ok) return packageRedirect('success');
  return packageRedirect(result.pending ? 'pending' : 'failed');
}

import { NextResponse } from 'next/server';
import { verifyAndActivateBankTransfer, verifyFlutterwaveWebhookHash } from '@/lib/billing/flutterwave';
export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';
export async function POST(request: Request) {
  if (!verifyFlutterwaveWebhookHash(request.headers.get('verif-hash')))
    return NextResponse.json({ error: 'Invalid webhook signature.' }, { status: 401 });
  const event = await request.json().catch(() => null) as
    { event?: string; data?: { id?: number; tx_ref?: string; payment_type?: string } } | null;
  if (!event || !event.data) return NextResponse.json({ error: 'Invalid webhook.' }, { status: 400 });
  if (event.event !== 'charge.completed' || event.data.payment_type !== 'bank_transfer')
    return NextResponse.json({ received: true });
  if (!event.data.tx_ref || event.data.id == null)
    return NextResponse.json({ error: 'Missing transaction identity.' }, { status: 400 });
  const result = await verifyAndActivateBankTransfer(event.data.tx_ref, undefined, event.data.id);
  if (result.ok) return NextResponse.json({ received: true, activated: true });
  if (result.pending) return NextResponse.json({ received: false }, { status: 503 });
  return NextResponse.json({ received: true, activated: false });
}

import { timingSafeEqual } from 'node:crypto';
import { createClient as createSupabaseClient } from '@supabase/supabase-js';

export function createBillingAdminClient() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !key) throw new Error('Supabase billing credentials missing.');
  return createSupabaseClient(url, key, { auth: { autoRefreshToken: false, persistSession: false } });
}

export function verifyFlutterwaveWebhookHash(received: string | null): boolean {
  const expected = process.env.FLW_SECRET_HASH;
  if (!expected || !received) return false;
  const a = Buffer.from(received);
  const b = Buffer.from(expected);
  return a.length === b.length && timingSafeEqual(a, b);
}

type PaymentResult =
  | { ok: true; alreadyPaid: boolean; companyId: string; planCode: string }
  | { ok: false; pending?: boolean; message: string };

type VerifiedTransaction = {
  id: number | string; tx_ref: string; amount: number; currency: string;
  status: string; payment_type?: string; created_at?: string;
};

export async function verifyAndActivateBankTransfer(
  reference: string, expectedCompanyId?: string, transactionId?: string | number,
): Promise<PaymentResult> {
  if (!/^JAAD-[a-f0-9]{32}$/i.test(reference)) return { ok: false, message: 'Invalid JAAD payment reference.' };
  let admin: ReturnType<typeof createBillingAdminClient>;
  try { admin = createBillingAdminClient(); }
  catch { return { ok: false, pending: true, message: 'Billing database is unavailable.' }; }
  const { data: attempt, error } = await admin.from('billing_payment_attempts')
    .select('id, company_id, plan_code, amount_kobo, currency, reference, status, payment_provider')
    .eq('reference', reference).maybeSingle();
  if (error || !attempt) return { ok: false, message: 'Payment reference was not found.' };
  if (attempt.payment_provider !== 'flutterwave') return { ok: false, message: 'Payment provider mismatch.' };
  if (expectedCompanyId && expectedCompanyId !== attempt.company_id) return { ok: false, message: 'Company mismatch.' };
  if (attempt.status === 'paid') return { ok: true, alreadyPaid: true, companyId: attempt.company_id, planCode: attempt.plan_code };
  if (attempt.status !== 'pending') return { ok: false, message: 'Payment is not pending.' };
  const secret = process.env.FLW_SECRET_KEY;
  if (!secret) return { ok: false, pending: true, message: 'Flutterwave is not configured.' };
  try {
    const endpoint = transactionId != null && /^\d+$/.test(String(transactionId))
      ? `https://api.flutterwave.com/v3/transactions/${encodeURIComponent(String(transactionId))}/verify`
      : `https://api.flutterwave.com/v3/transactions/verify_by_reference?tx_ref=${encodeURIComponent(reference)}`;
    const response = await fetch(endpoint, { headers: { Authorization: `Bearer ${secret}` }, cache: 'no-store' });
    const payload = await response.json().catch(() => null) as { status?: string; data?: VerifiedTransaction } | null;
    if (!response.ok || payload?.status !== 'success' || !payload.data)
      return { ok: false, pending: true, message: 'Flutterwave verification is not available yet.' };
    const tx = payload.data;
    if (tx.status !== 'successful') return { ok: false, pending: true, message: 'Payment is not confirmed.' };
    if (tx.tx_ref !== attempt.reference || tx.currency !== 'NGN' || attempt.currency !== 'NGN' ||
        Number(tx.amount) * 100 !== Number(attempt.amount_kobo))
      return { ok: false, message: 'Payment reference, currency or amount mismatch.' };
    if (tx.payment_type !== 'bank_transfer') return { ok: false, message: 'Only verified bank transfers are accepted.' };
    const { error: activationError } = await admin.rpc('jaad_activate_subscription_from_payment', {
      p_reference: reference,
      p_paystack_transaction_id: String(tx.id), // Legacy SQL parameter name; value is Flutterwave transaction ID.
      p_paid_at: tx.created_at || new Date().toISOString(),
    });
    if (activationError) return { ok: false, pending: true, message: 'Payment confirmed, subscription activation pending.' };
    return { ok: true, alreadyPaid: false, companyId: attempt.company_id, planCode: attempt.plan_code };
  } catch {
    return { ok: false, pending: true, message: 'Unable to verify payment with Flutterwave.' };
  }
}
